## 标题 字符串匹配

时间限制：1000 ms $\quad$ 空间限制：512 MiB

```cpp
#include <iostream>
#include <cstring>
using namespace std;
int n, m, C;
int s[1000005]; 
int t[1000005];
int lst[1000005];
int fail[1000005];
int ans[1000005];
int x, tot;

/*
1 3
5 5
1 1 3 1 3
2 2 3 1 3
*/
void solve() {
	tot = 0;
	memset(lst, -1, sizeof(lst));
	memset(s, 0, sizeof(s));
	memset(t, 0, sizeof(t));
	scanf("%d %d", &n, &m);
	for (int i = 0; i < n; i++) {
		scanf("%d", &x);
		if (~lst[x]) s[i] = i - lst[x];
		lst[x] = i;
	}
	
	memset(lst, -1, sizeof(lst));
	for (int i = 0; i < m; i++) {
		scanf("%d", &x);
		if (~lst[x]) t[i] = i - lst[x];
		lst[x] = i;
	}
	t[m] = -1;
	
	for (int i = 1; i < m; i++) {
		int j = fail[i-1];
		while (j > 0 && t[j] && t[i] != t[j]) j = fail[j-1];
		if (t[j] == 0 || t[i] == t[j]) j++;
		fail[i] = j;
	}
	
	int pre = 0;
	for (int i = 0; i < n; i++) {
		int j = pre;
		while (j > 0 && t[j] && s[i] != t[j]) j = fail[j-1];
		if (t[j] == 0 || s[i] == t[j]) j++;
		if (j == m) ans[++tot] = i + 2 - m;
		pre = j;
	}
	
	printf("%d\n", tot);
	for (int i = 1; i <= tot; i++) printf("%d ", ans[i]);
	printf("\n");
}
int main() {
	int t;
	scanf("%d %d", &t, &C);
	while (t--)
		solve(); 
	return 0;
}
```
